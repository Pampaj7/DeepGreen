% Online Resource 1 (esm.tex, built beside this file) carries the detail cut
% from the article. Its section, table and figure numbers come from its own

% is a resolved reference and not typed text. paper/build.sh compiles main,
% then esm, then main again; the submission zip ships esm.aux beside main.tex

% disjoint, so no prefix is needed.

\newcommand{\esmfig}[1]{Online Resource~1, Fig.~\ref{#1}}

% The tables were set for cas-dc's two-column page, whose table* spans about
% 505 pt; sn-jnl's single column is 372 pt. sn-jnl sets its own table
% environment in \tablebodyfont (8 bp) but leaves table* at the body size, so
% here every table float -- starred or not -- gets the class's table font and a
% tighter column gap: one type-size choice for the whole EMSE layout rather
% than a \resizebox per table, which would set each at a different size. It
% goes into \@floatboxreset because that resets the font after any
% environment hook has run.
\makeatletter
\g@addto@macro\@floatboxreset{%
  \def\@tempa{table}%
  \ifx\@captype\@tempa\tablebodyfont\setlength{\tabcolsep}{4pt}\fi}
\makeatother

\input{\paperroot preamble}
% Imported after the preamble: esm.aux carries captions that use its macros.
\usepackage{xr-hyper}
% xr-hyper is loaded after hyperref (the class loads it), so hyperref has not
% installed its five-field label import; install it here, keeping each field
% unexpanded so that captions using \si{\second} or the number macros are
% stored as written rather than expanded in the preamble.
\makeatletter
\def\XR@addURL#1{\XR@@dURL#1{}{}{}{}\\}
\def\XR@@dURL#1#2#3#4#5\\{%
  {\unexpanded{#1}}{\unexpanded{#2}}%
  \if!#4!\else{\unexpanded{#3}}{#4}{\XR@URL}\fi}
% Remote links carry a named destination, so the page hyperref asks for is
% unused; reset it per link, or every link after the first warns that page 0
% is not a page.
\let\DG@MakeRemoteAction\Hy@MakeRemoteAction
\def\Hy@MakeRemoteAction{\let\Hy@href@page\@empty\DG@MakeRemoteAction}
\makeatother
