% xr-hyper is loaded after hyperref (the class loads it), so hyperref has not
% installed its five-field label import; install it here, keeping each field
% unexpanded so that captions using \si{\second} or the number macros are
% stored as written rather than expanded in the preamble.
\makeatletter
\def\XR@addURL#1{\XR@@dURL#1{}{}{}{}\\}
\def\XR@@dURL#1#2#3#4#5\\{%
  {\unexpanded{#1}}{\unexpanded{#2}}%
  \if!#4!\else{\unexpanded{#3}}{#4}{\XR@URL}\fi}
% Remote links carry a named destination, so the page hyperref asks for is
% unused; reset it per link, or every link after the first warns that page 0
% is not a page.
\let\DG@MakeRemoteAction\Hy@MakeRemoteAction
\def\Hy@MakeRemoteAction{\let\Hy@href@page\@empty\DG@MakeRemoteAction}
\makeatother

\noindent This document is Online Resource~1 of the article \emph{Deep Green
AI: Energy Efficiency of Deep Learning across Language--Framework Ecosystems}.
It carries the material the article condenses or leaves out: the related-work
comparison table, the full experiment specification and execution detail,
supplementary analyses for RQ1, RQ3 and RQ4 and for the accelerator-saturation
cell, an illustrative industrial scenario, the complete catalogue of defect
classes, and the extended threats to validity. Section, table and figure
numbers without a letter (Section~\ref{subsec:rq3}, Table~\ref{tab:spread})
refer to the article; lettered ones (Section~\ref{app:collapse},
Table~\ref{tab:precision_contrast}) refer to this document, and ``this paper''
in the text below means the article. Every quantity is
generated by the same analysis pipeline as the article's and injected at build
time.
